2. Form Design (Default TextBox Names)To prevent the ORA-00936 error caused by mismatched names, 
use the default sequence when dragging controls onto the form.
 TextBox1: Student ID  
 TextBox2: Student Name  
 TextBox3: OS Mark  
 TextBox4: RDBMS Mark   
 TextBox5: CN Mark  
 TextBox6: SE Mark  
 TextBox7: Total Marks   
 TextBox8: Result (PASS/FAIL)  
 TextBox9: Percentage  
Buttons: Button1 (Calculate), Button2 (Insert), Button3 (Delete), Button4 (View)
DataGridView1: placed at the bottom for the View Report functionality. btnViewDataGridView1: Placed at the bottom for the View Report functionality.


3.form 
```vb
Imports System.Data.OleDb

Public Class Form1
    Dim cn As New OleDbConnection("Provider=MSDAORA;Data Source=XE;User ID=system;Password=system;")

    ' --- CALCULATE ---
    Private Sub Button1_Click(sender As Object, e As EventArgs) Handles Button1.Click
        TextBox7.Text = (Val(TextBox3.Text) + Val(TextBox4.Text) + Val(TextBox5.Text) + Val(TextBox6.Text)).ToString()

        If Val(TextBox3.Text) <= 30 Or Val(TextBox4.Text) <= 30 Or Val(TextBox5.Text) <= 30 Or Val(TextBox6.Text) <= 30 Then
            TextBox8.Text = "FAIL"
        Else
            TextBox8.Text = "PASS"
        End If

        TextBox9.Text = (Val(TextBox7.Text) / 4).ToString()
    End Sub

    ' --- INSERT ---
    Private Sub Button2_Click(sender As Object, e As EventArgs) Handles Button2.Click
        Button1.PerformClick()   ' auto-calculate before insert
        Try
            cn.Open()
            Dim sql As String = "INSERT INTO st VALUES (" & TextBox1.Text & ", '" & TextBox2.Text & "', " & TextBox3.Text & ", " & TextBox4.Text & ", " & TextBox5.Text & ", " & TextBox6.Text & ", " & TextBox7.Text & ", '" & TextBox8.Text & "', " & TextBox9.Text & ")"
            Dim cmd As New OleDbCommand(sql, cn)
            cmd.ExecuteNonQuery()

            MsgBox("Successfully Inserted")

            TextBox1.Clear(): TextBox2.Clear(): TextBox3.Clear()
            TextBox4.Clear(): TextBox5.Clear(): TextBox6.Clear()
            TextBox7.Clear(): TextBox8.Clear(): TextBox9.Clear()
        Catch ex As Exception
            MsgBox(ex.Message)
        Finally
            cn.Close()
        End Try
    End Sub

    ' --- DELETE ---
    Private Sub Button3_Click(sender As Object, e As EventArgs) Handles Button3.Click
        Dim studentID As String = InputBox("Enter the Student ID to delete:")
        If studentID = "" Then Exit Sub

        Try
            cn.Open()
            Dim cmd As New OleDbCommand("DELETE FROM st WHERE stid = " & studentID, cn)
            Dim rowsAffected As Integer = cmd.ExecuteNonQuery()

            If rowsAffected > 0 Then
                MsgBox("Record with ID " & studentID & " deleted successfully!")
            Else
                MsgBox("No record found with ID: " & studentID)
            End If
        Catch ex As Exception
            MsgBox(ex.Message)
        Finally
            cn.Close()
        End Try
    End Sub

    ' --- VIEW ---
    Private Sub Button4_Click(sender As Object, e As EventArgs) Handles Button4.Click
        Try
            Dim da As New OleDbDataAdapter("SELECT * FROM st", cn)
            Dim dt As New DataTable()
            da.Fill(dt)
            DataGridView1.DataSource = dt
        Catch ex As Exception
            MsgBox(ex.Message)
        End Try
    End Sub
End Class
```
