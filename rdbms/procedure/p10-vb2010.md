**10th program 

1.sql 

```sql
CREATE TABLE emp (
    empid NUMBER PRIMARY KEY,
    empname VARCHAR2(50),
    salary NUMBER(10,2),
    gross NUMBER(10,2),
    net NUMBER(10,2)
);
```

2. Form Design (Default TextBox Names)
 TextBox1: Employee ID  
 TextBox2: Employee Name  
 TextBox3: Basic Salary  
 TextBox4: Total Allowance (Calculated)  
 TextBox5: Total Deduction (Calculated)  
 TextBox6: Gross Salary (Calculated)  
 TextBox7: Net Salary (Calculated)  
 Buttons: btnCalculate, btnInsert, btnDelete, btnView,  btnExit
 DataGridView1: Placed at the bottom.

3.vb code 
```vb
Imports System.Data.OleDb

Public Class Form10
    Dim cn As New OleDbConnection("Provider=MSDAORA;Data Source=XE;User ID=system;Password=system;")

    ' --- CALCULATE PAYROLL ---
    Private Sub btnCalculate_Click(sender As Object, e As EventArgs) Handles btnCalculate.Click
        ' Allowance = Salary * 0.25
        TextBox4.Text = (Val(TextBox3.Text) * 0.25).ToString()
        
        ' Deduction = Salary * 0.12
        TextBox5.Text = (Val(TextBox3.Text) * 0.12).ToString()
        
        ' Gross = Salary + Allowance
        TextBox6.Text = (Val(TextBox3.Text) + Val(TextBox4.Text)).ToString()
        
        ' Net = Gross - Deduction
        TextBox7.Text = (Val(TextBox6.Text) - Val(TextBox5.Text)).ToString()
    End Sub

    ' --- INSERT RECORD ---
    Private Sub btnInsert_Click(sender As Object, e As EventArgs) Handles btnInsert.Click
        Try
            cn.Open()
            ' Only inserting the 5 fields defined in the MS Access table structure
            Dim sql As String = "INSERT INTO emp VALUES (" & TextBox1.Text & ", '" & TextBox2.Text & "', " & TextBox3.Text & ", " & TextBox6.Text & ", " & TextBox7.Text & ")"
            Dim cmd As New OleDbCommand(sql, cn)
            cmd.ExecuteNonQuery()
            
            MsgBox("Successfully Inserted")
            
            TextBox1.Clear(): TextBox2.Clear(): TextBox3.Clear()
            TextBox4.Clear(): TextBox5.Clear(): TextBox6.Clear()
            TextBox7.Clear()
        Catch ex As Exception
            MsgBox(ex.Message)
        Finally
            cn.Close()
        End Try
    End Sub

    ' --- DELETE RECORD ---
    Private Sub btnDelete_Click(sender As Object, e As EventArgs) Handles btnDelete.Click
        Dim eid As String = InputBox("Enter the EMPLOYEE ID to delete:")
        If eid = "" Then Exit Sub

        Try
            cn.Open()
            Dim sql As String = "DELETE FROM emp WHERE empid = " & eid
            Dim cmd As New OleDbCommand(sql, cn)
            Dim rowsAffected As Integer = cmd.ExecuteNonQuery()
            
            If rowsAffected > 0 Then
                MsgBox("Record with ID " & eid & " deleted successfully!")
            Else
                MsgBox("No record found with ID: " & eid)
            End If
        Catch ex As Exception
            MsgBox(ex.Message)
        Finally
            cn.Close()
        End Try
    End Sub

    ' --- VIEW REPORT ---
    Private Sub btnView_Click(sender As Object, e As EventArgs) Handles btnView.Click
        Try
            Dim da As New OleDbDataAdapter("SELECT * FROM emp", cn)
            Dim dt As New DataTable()
            da.Fill(dt)
            DataGridView1.DataSource = dt
        Catch ex As Exception
            MsgBox(ex.Message)
        End Try
    End Sub

    ' --- EXIT ---
    Private Sub btnExit_Click(sender As Object, e As EventArgs) Handles btnExit.Click
        Application.Exit()
    End Sub
End Class
```
