```sql
    salary  NUMBER,
    gross   NUMBER,

);
```

```vb
Imports System.Data.OleDb

Public Class Form1
    Dim cn As New OleDbConnection("Provider=MSDAORA;Data Source=XE;User ID=system;Password=system;")

    ' --- CALCULATE ---
    Private Sub Button1_Click(sender As Object, e As EventArgs) Handles Button1.Click
        TextBox4.Text = (Val(TextBox3.Text) * 0.25).ToString()                    ' Allowance
        TextBox5.Text = (Val(TextBox3.Text) * 0.12).ToString()                    ' Deduction
        TextBox6.Text = (Val(TextBox3.Text) + Val(TextBox4.Text)).ToString()      ' Gross
        TextBox7.Text = (Val(TextBox6.Text) - Val(TextBox5.Text)).ToString()      ' Net
    End Sub

    ' --- INSERT ---
    Private Sub Button2_Click(sender As Object, e As EventArgs) Handles Button2.Click
        Button1.PerformClick()   ' auto-calculate before insert
        Try
            cn.Open()
            Dim sql As String = "INSERT INTO emp VALUES (" & TextBox1.Text & ", '" & TextBox2.Text & "', " & TextBox3.Text & ", " & TextBox6.Text & ", " & TextBox7.Text & ")"
            Dim cmd As New OleDbCommand(sql, cn)
            cmd.ExecuteNonQuery()

            MsgBox("Successfully Inserted")

            TextBox1.Clear(): TextBox2.Clear(): TextBox3.Clear()
            TextBox4.Clear(): TextBox5.Clear(): TextBox6.Clear()
            TextBox7.Clear()
        Catch ex As Exception
            MsgBox(ex.Message)
        Finally
            cn.Close()
        End Try
    End Sub

    ' --- DELETE ---
    Private Sub Button3_Click(sender As Object, e As EventArgs) Handles Button3.Click
        Dim eid As String = InputBox("Enter the EMPLOYEE ID to delete:")
        If eid = "" Then Exit Sub

        Try
            cn.Open()
            Dim cmd As New OleDbCommand("DELETE FROM emp WHERE empid = " & eid, cn)
            Dim rowsAffected As Integer = cmd.ExecuteNonQuery()

            If rowsAffected > 0 Then
                MsgBox("Record with ID " & eid & " deleted successfully!")
            Else
                MsgBox("No record found with ID: " & eid)
            End If
        Catch ex As Exception
            MsgBox(ex.Message)
        Finally
            cn.Close()
        End Try
    End Sub

    ' --- VIEW ---
    Private Sub Button4_Click(sender As Object, e As EventArgs) Handles Button4.Click
        Try
            Dim da As New OleDbDataAdapter("SELECT * FROM emp", cn)
            Dim dt As New DataTable()
            da.Fill(dt)
            DataGridView1.DataSource = dt
        Catch ex As Exception
            MsgBox(ex.Message)
        End Try
    End Sub

    ' --- EXIT ---
    Private Sub Button5_Click(sender As Object, e As EventArgs) Handles Button5.Click
        Application.Exit()
    End Sub
End Class
```
